CREATE TABLE `documents` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `project_id` int(11) NOT NULL,
  `progress_report_id` int(11) DEFAULT NULL,
  `document_type_id` int(11) NOT NULL,
  `title` varchar(255) NOT NULL,
  `file_path` text NOT NULL,
  `file_name` varchar(255) NOT NULL,
  `file_size` int(11) DEFAULT NULL,
  `uploaded_by` int(11) NOT NULL,
  `is_locked` tinyint(1) DEFAULT 0,
  `created_at` timestamp NOT NULL DEFAULT current_timestamp(),
  `document_type` varchar(100) NOT NULL DEFAULT 'other_report',
  `url` text DEFAULT NULL,
  `uploaded_at` timestamp NOT NULL DEFAULT current_timestamp(),
  `status` varchar(20) DEFAULT 'pending',
  `action_required` text DEFAULT NULL,
  `action_assignee_id` int(11) DEFAULT NULL,
  `action_status` varchar(20) DEFAULT 'pending',
  `action_response` text DEFAULT NULL,
  `description` text DEFAULT NULL,
  `size` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `progress_report_id` (`progress_report_id`),
  KEY `uploaded_by` (`uploaded_by`),
  KEY `idx_documents_project` (`project_id`),
  KEY `idx_documents_type` (`document_type_id`),
  CONSTRAINT `documents_ibfk_1` FOREIGN KEY (`project_id`) REFERENCES `projects` (`id`) ON DELETE CASCADE,
  CONSTRAINT `documents_ibfk_2` FOREIGN KEY (`progress_report_id`) REFERENCES `progress_reports` (`id`) ON DELETE CASCADE,
  CONSTRAINT `documents_ibfk_3` FOREIGN KEY (`document_type_id`) REFERENCES `document_types` (`id`),
  CONSTRAINT `documents_ibfk_4` FOREIGN KEY (`uploaded_by`) REFERENCES `users` (`id`)
) ENGINE=InnoDB AUTO_INCREMENT=3 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci
